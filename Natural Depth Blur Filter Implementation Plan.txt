Implementation Plan: Natural Depth Blur Filter (Optical Bokeh Simulation)
Objective
Recreate natural optical depth-of-field blur in Android instead of modern harsh semantic segmentation cutouts. Simulate physical camera optics using continuous monocular depth estimation and a custom Circle of Confusion (CoC) bokeh shader.
1. Core Architecture & Pipeline
Depth Estimation over Binary Masks
* Integrate a lightweight monocular depth estimation model (e.g., Depth Anything v2 Small or MiDaS v2.1 quantized to TFLite).
* Generate an 8-bit or 16-bit continuous grayscale depth map rather than a binary mask.
Focal Plane & Depth of Field (DoF)
* Tap-to-focus sets target focal depth d0 (0.0 to 1.0).
* Define a DoF in-focus zone thickness where blur radius is 0.
* Calculate blur radius proportionally based on Circle of Confusion distance from d0.
Bleed Prevention & Bokeh Sampling
* Implement depth-weighted sampling in the fragment shader to prevent background color from leaking into sharp foreground edges.
* Use a Poisson disc sampling kernel (circular aperture simulation) rather than Gaussian box blur.
2. Shader Implementation (AGSL / RuntimeShader)
uniform shader image;


uniform shader depthMap;


uniform float2 resolution;


uniform float focalDepth;      // 0.0 to 1.0 (depth at focus point)


uniform float dofThickness;    // e.g., 0.05


uniform float maxBlurRadius;   // e.g., 16.0


uniform float falloff;         // e.g., 0.25


const int SAMPLES = 16;


const float2 POISSON[16] = float2[16](


    float2(-0.326, -0.406), float2(-0.840, -0.074),


    float2(-0.696,  0.457), float2(-0.203,  0.621),


    float2( 0.962, -0.195), float2( 0.473, -0.480),


    float2( 0.519,  0.767), float2( 0.185, -0.893),


    float2( 0.507,  0.064), float2( 0.896,  0.412),


    float2(-0.322, -0.933), float2(-0.792, -0.598),


    float2(-0.301,  0.154), float2( 0.399,  0.374),


    float2( 0.023, -0.222), float2( 0.035,  0.342)


);


half4 main(float2 fragCoord) {


    float2 uv = fragCoord / resolution;


    


    // Depth at current pixel (grayscale 0.0 to 1.0)


    float currentDepth = depthMap.eval(fragCoord).r;


    


    // Calculate circle of confusion


    float depthDiff = abs(currentDepth - focalDepth);


    float blurFactor = clamp((depthDiff - dofThickness) / falloff, 0.0, 1.0);


    float radius = blurFactor * maxBlurRadius;


    


    if (radius < 0.5) {


        return image.eval(fragCoord);


    }


    half4 accumColor = half4(0.0);


    float totalWeight = 0.0;


    for (int i = 0; i < SAMPLES; i++) {


        float2 sampleCoord = fragCoord + POISSON[i] * radius;


        


        // Sample depth at neighbor to prevent background bleeding onto foreground


        float sampleDepth = depthMap.eval(sampleCoord).r;


        float weight = 1.0;


        


        // Attenuate weight if center is background and sample is foreground


        if (currentDepth > focalDepth && sampleDepth < currentDepth - 0.1) {


            weight = 0.1;


        }


        accumColor += image.eval(sampleCoord) * weight;


        totalWeight += weight;


    }


    return accumColor / totalWeight;


}
3. Kotlin Wrapper Class
class NaturalBokehEffect(context: Context) {


    private val shader = RuntimeShader(bokehShaderSource)


    fun apply(


        sourceBitmap: Bitmap, 


        depthBitmap: Bitmap, 


        focusDepth: Float, 


        intensity: Float = 16f


    ): Bitmap {


        val output = Bitmap.createBitmap(sourceBitmap.width, sourceBitmap.height, Bitmap.Config.ARGB_8888)


        val canvas = Canvas(output)


        val imageShader = BitmapShader(sourceBitmap, Shader.TileMode.CLAMP, Shader.TileMode.CLAMP)


        val depthShader = BitmapShader(depthBitmap, Shader.TileMode.CLAMP, Shader.TileMode.CLAMP)


        shader.setInputShader("image", imageShader)


        shader.setInputShader("depthMap", depthShader)


        shader.setFloatUniform("resolution", sourceBitmap.width.toFloat(), sourceBitmap.height.toFloat())


        shader.setFloatUniform("focalDepth", focusDepth)


        shader.setFloatUniform("dofThickness", 0.04f)


        shader.setFloatUniform("maxBlurRadius", intensity)


        shader.setFloatUniform("falloff", 0.20f)


        val paint = Paint().apply { this.shader = shader }


        canvas.drawRect(0f, 0f, sourceBitmap.width.toFloat(), sourceBitmap.height.toFloat(), paint)


        return output


    }


}
